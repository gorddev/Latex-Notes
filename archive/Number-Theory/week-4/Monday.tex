\subsection*{Pythagorean Triple}
First we begin by discussing Pythagorean Triples:
\begin{definition}{Pythagorean Triple}{}\gline
  A triple of integers $(x,y,z)$ is called a \ul{Pythagorean Triple} if:
  \begin{align*}
    x^2 + y^2 = z^2
  \end{align*}
  For example, $(3, 4, 5)$ is a Pythagorean Triple.
\end{definition}
There's also an important relationship between scalar multiples of Pythagorean triples.
\begin{theorem}{Scalars of Pythagorean Triples}{}\gline
  Let $d>0$. Then $(x,y,z)$ is a \ul{Pythagorean Triple} if and only if $(dx, dy, dz)$ is a Pythagorean Triple.
  \begin{corollary}{}{}\gline
  If any two of $x,y,z$ are divisible by $d$, then all three of $x,y,z$ are divisible by $d$.
\end{corollary}
\end{theorem}

\begin{definition}{Primitive Pythagorean Triple}{}\gline
  A Pythagorean Triple $(x,y,z)$ where $\gcd(x,y,z)=1$ is called a \ul{primitive Pythagorean Triple}.
  \begin{criterion}{Primitive Properties}{}\gline
      All Primitives of the form $x^2+y^2 = z^2$ must have the following properties:
      \begin{itemize}[itemsep=0pt, parsep=0pt]
        \item One of $x^2, y^2$ is $0\gmod{4}$
        \item One of $x^2,y^2$ is $1\gmod{4}$
        \item $z^2 \equiv 1\gmod 4$. 
      \end{itemize}
  \end{criterion}
\end{definition}
