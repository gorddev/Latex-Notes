\section*{1.5 - Powers and Roots}
\begin{definition}{The $n$-th roots of unit}{}
  \begin{align*}
    \omega{n_k}=e^{i\frac{2\pi k}n}\quad k = 0,1,\ldots, n-1
  \end{align*}
  are the solutions to $z^n = 1$. We define:
  \begin{align*}
    \omega_n = \omega_{n,1}=e^{i\frac{2\pi}n}
  \end{align*}
  to be the \ul{principle $n^\text{th}$ root of unity}.
\end{definition}
\newcommand{\enth}{$n^\text{th}$} The \enth roots of unity:
\begin{itemize}
  \item are the verteices of a regular $n-$gon inscribed in the unit circle.
  \item if non-real, they come in conjugate pairs (if $z_0$ is a solution to $z^n = 1$, so is $\bar{z_0}$)
  \item can be written $\omega_{n,k}=(\omega_n)^k$, where $\omega_n$ is the principal one. 
\end{itemize}
\subsection*{General Roots}
Let $z = re^{i\theta}$. Say we want the solutions to ${S}^n = z$.
\begin{itemize}
  \item The \ul{principal \enth root of $z$ is}:
  \begin{align*}
    S = \sqrt[n]r e^{i\frac{\Arg{z}}n} = r^{\frac{1}n}e^{i\frac{\Arg{z}}n}
  \end{align*}
  \item The whole set of \enth roots of $z$ are:
  \begin{align*}
    S = r^{\frac{1}n}e^{i\cpr{\frac{\theta+2\pi k}n}}\quad k = 0,1,\ldots, n -1
  \end{align*}
  Note 1: We can think of this four ways.
  \begin{itemize}
    \item Integral of taking the \enth root of $4re^{i\theta}$, take hte \enth root of $re^{i\theta + 2\pi k}$:
    \begin{align*}
      S = r^{\frac{1}n}e^{i\cpr{\frac{\theta+2\pi k}n}}
    \end{align*}
  \end{itemize}
\end{itemize}
