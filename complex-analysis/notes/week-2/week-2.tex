\subsection{The Algebra of Complex Numbers}
First, we cover the basic sets of numbers
\begin{align*}
  \N &= \{0, 1, 2\}\\
  \Z &= \{-2, -1, 0, 1, 2\}\\
  \Q &= \{\frac{a}b \mid a\in \Z, b\in \N \backslash\{0\}\}\\
  \R &= \text{ Solving Equations like } x^2 - 2 = 0\\
  \C &= \{a+bi\mid a,b\in\R\}
\end{align*}
\begin{definition}{Complex Numbers}{}\gline{}
  Denote by $i$ the number $\sqrt{-1}$. We define the \ul{set of complex numbers}
  \begin{align*}
    \C =\{a+bi\mid a,b\in\R\}
  \end{align*}
  \centerit{\ul{\textbf{Equality}}}
  Two complex numbers $a+bi$ and $c+di$ are said to be equal if and only if:
  \begin{align*}
    a = c \text{ and } b = d
  \end{align*}
  \centerit{\ul{\textbf{Addition}}}
  We add complex numbers with the following formula;
  \begin{gather*}
    (a+bi) + (c + di) = (a+c) + (c + d)i
  \end{gather*}
  \centerit{\ul{\textbf{Multiplication}}}
  We multiply complex numbers with the following formula;
  \begin{align*}
    (a+bi)(c+di)&= (ac + adi + bci + bdi^2)\\
    &= (ac-bd) + (ad+bc)i
  \end{align*}
\end{definition}
We will cover all the axioms of $\C$ at a later point. 
\begin{theorem}{$\C$ is a field}{}\gline{}
  \textit{Note: }$\C$ is not an ordered field, meaning that there is no '$<$' relation.
\end{theorem}
Now, we work on several different problems:
\begin{enumerate}
  \item Express the following complex numbers for $a+ bi$.
  \begin{enumerate}
    \item $(a+3i)+(4-i)$
    \[(4 + a)-2i\]
    \item $(2+3i)(4-i)$
    \begin{gather*}
      8 + 3 - 2i + 12 i\\
      11 + 10i
    \end{gather*}
    \item $\displaystyle\frac{2+3i}{4-i}$
    \begin{align*}
      \frac{(2+3i)}{4-i}&= \frac{(2+3i)(4+i)}{(4-i)(4+i)}\\
      &= \left(\frac{5 + 14i}{15}\right)
    \end{align*}
  \end{enumerate}
  \item Find all solutions to $x^2 = 3 + 4i$. 
\end{enumerate}
\begin{definition}{Complex Conjugate}{}\gline{}
  The \ul{complex conjugate} of $z = a+bi$ is the number $\bar z = a - bi$. 
\end{definition}

\begin{definition}{Real \& Imaginary Part}{}\gline{}
  The \ul{real part} of $z = a+bi$ is $\real{z} = a$. \\
  The \ul{imaginary part} of $z = a+bi$ is $\im{z} = b$. \gline{}
  \textit{Note: }$\real{z},\im{z}\in \R$. 
\end{definition}

\begin{enumerate}
  \item [2.] Find all solutions to $z^2 = 3+ 4i$.
  \begin{problem}
    We want to use the formula $z = (a+bi)$. Thus, we get that:
    \begin{align*}
      z^2 = a^2 + 2abi - b^2
    \end{align*}
    Then, we find that:
    \begin{align*}
      3&= a^2 - b^2\\
      4 &= 2(ab) \quad\rightarrow \quad 2 &= (ab)
    \end{align*}
    So we get the solutions $(a,b) = (2, 1), (-2, -1)$.
  \end{problem}
  \item [3.] Find all solutions to $x^3 = 1$. 
  \begin{problem}
    We represent $z$ as $z = (a+bi)$. Thus, we find that:
    \begin{align*}
      z^3 &= a^3 + 3 a^2 b i + 3 a b^2 i^2 + b^3 i^3\\
      &= a^3 - 3ab + 3a^2bi - b^3 i
    \end{align*}
    And we get the following system of equations:
    \begin{align*}
      a^3 - 3ab &= 1\\
      3a^2b - b^3 &= 0
    \end{align*}
    For $a, b\ne 0$, we find that $3a^2 = b^2$. WLOG $a = b$ or $a = -b$. 
  \end{problem}
\end{enumerate}

There is a natural bijection between $\C$ and the $xy$-coordinate system $\R^2$:
\begin{align*}
  a + bi \leftrightarrow (a,b)
\end{align*}
\begin{theorem}{Properties of Complex Nums}{}
  \begin{enumerate}
    \item $\gbar{ x \pm w} = \bar z \pm \bar w$
    \item $\gbar{zw} = \bar z \cdot \bar w$
    \item $\gbar{\cpr{\frac{z}w}} = \frac{\bar z}{\bar w}$
    \item $\Re{z} = \frac{z + \bar z}{z}$
    \item $\Im{z}= \frac{z - \bar z}{zi}$
  \end{enumerate}
\end{theorem}

We saw last time that complex addition behaves like vector addition.\gline{}
Now, consider \ul{Polar Form}:

More formalized, for \textit{any $z\in\C$ of the form $a + bi$}, we can write:
\begin{align*}
  z = &= a + bi\\
  &= r(\cos\theta +i\sin\theta)\\
  &= r\cis\theta
\end{align*}
In this case, we write $r = \sqrt{a^2 + b^2}=|z|$. Note that $\theta$ is defined as counter-clockwise from the positive real axis. \gline{}
Here's some quick conversion equations:
\begin{align*}
  \mathrm{Polar}&\to\mathrm{Cartesian}\\
  x &= r\cos\theta\\
  y&= r\sin\theta\\
  \mathrm{Cartesian}&\to\mathrm{Polar}\\
  r &= \sqrt{x^2 + y^2}\\
  \theta &= \tan^{-1}\cpr{\frac{y}x}^*
\end{align*}
\textit{*Note: If $\theta$ is actually in the 2nd or 3rd quadrant, you'll need to add $\pi$ to $\tan^{-1}\cpr{\frac{y}x}$ to get a valid angle for $\theta$}\gline{}
\begin{definition}{Argument of $z$}{}\gline{}
  The \ul{argument of $z$}, denoted $\arg(z)$, is the set ofall positive values of $\theta$ in the polar form of $z$. Note that $\arg(z)$ is undefined for $z = 0$.  
  \begin{example}
    Let $z = -3 -3i$. Then, $\arg(z)=\ccbr{\frac{5\pi}4 + 2\pi k\mid k\in \Z}$.
  \end{example}
\end{definition}
\begin{definition}{Principal Argument of Z}{}\gline{}
  The \ul{principle oargument of $z$}, denoted $\Arg{z}$, is the element of $\arg(z)$ whose value is in the interval $(-\pi, \pi]$.
  \begin{corollary}{Branch of $\arg(z)$}{}\gline{}
    The notation $\arg_\tau(z)$ will denote the element of $\arg(z)$ whose value lies in the interval $(\tau, \tau + 2\pi]$. This is called a \ul{branch} of $\arg(z)$. 
  \end{corollary}
\end{definition}

\subsection{The Complex Exponential}
First, we begin with Euler's formula and how we can formulate powers of imaginary numbers. 
\begin{definition}{Euler's Formula}{}\gline{}
  Given $y\in\R$, we define $e^{iy}$ to be:
  \begin{align*}
    e^{iy}=\cos y + i \sin y \text{ (Euler's Formula)}
  \end{align*}
  More generally, for $z = x + iy$, we define:
  \begin{align*}
    e^{z}&= e^x(\cos y + i\sin y)
  \end{align*}
\end{definition}
Note that this function agrees with the formula for real-valued inputs, because if $iy = 0$, then $e^z = e^x$. If you remember Taylor or Maclauren series, you may note that the expansion of $e^x$ is deeply related to sines and cosines, and this is that relationship embodied. \gline{}
This is also a much shorter way to use polar coordinates. \gline{}From now on, we'll write the polar form of any given $z$ as:
\begin{align*}
  x = re^{i\theta}, \text{ where }r = |z| \text{ and }\theta\in\arg(z)
\end{align*}
\begin{example}
  Let $z = \sqrt 3 + i$. Then:
  \begin{align*}
    e^z = e^{\sqrt 3 + 1} &= e^{\sqrt 3}\cdot e^i\\
    &\approx 5.652\cdot e^{i}
  \end{align*}
  So, $e^{\sqrt 3 + i}$ has $r = e^{\sqrt 3}$ and $\theta = 1$. 
\end{example}
We can formulate some general properties for multiplication as well. Consider this formulation:
\begin{align*}
  z &= re^{i\theta} \text{ and } w= se^{i\phi}, \text{ then}\\
  zw&= (re^{i\theta}(se^{i\phi}))\\
  &= rse^{i\theta + \phi}\ahline
  \frac{z}w &= \frac{r}se^{i(\theta - \phi)}\\
  z^n&= r^ne^{in\theta}\text{, for }n\in\Z\\
  \frac{1}z&= \frac{1}re^{-i\theta}
\end{align*}

\subsection{Planar Sets}
\begin{definition}{Open Disk}{}\gline{}
  Given $z\in\C$ and $r>0$, the \ul{open disk} of radius $r$ centered at $z$ is:
  \begin{align*}
    D(z, r)=D_r(z)=D_z(r)=\{w\in\C\mid |w-z|<r\}
  \end{align*}
  This is also called a (circular) neighborhood of $z$. 
\end{definition}

\begin{definition}{Interior Point}{}\gline{}
  A point $z\in A$ is an \ul{interior point} of $A$ if there exists $\varepsilon > 0$ such that $D_\varepsilon(z) \subseteq A$.
\end{definition}
\begin{definition}{Exterior Point}{}\gline{}
  A point $z\in\C$ is an \ul{exterior point} of $A$ if there exists $\varepsilon > 0$ such that $D_\varepsilon(z)\subseteq \C\backslash A$, the complement of $A$.
\end{definition}
\begin{definition}{Boundary Point}{}\gline{}
  A point $z\in\C$ is a \ul{boundary point} of $A$ if $z$ is neither an interior point nor an exterior point of $A$. 
  \begin{theorem}{Boundary Point Cont.}{}\gline{}
    $z$ is a boundary point of $A$ if and only if for all $\varepsilon > 0$, $D_\varepsilon(z)$ contians points in $A$ and points in $\C\backslash A$.
  \end{theorem}
\end{definition}
\begin{definition}{Open Set}{}\gline{}
  A set $A\subseteq\C$ is \ul{open} if every $z\in A $ is an interior point of $A$.\gline{}
  \textit{Note:} Equivalently, $A$ is open if $\forall z\in A, \exists\varepsilon > 0$ such that $D_\varepsilon(z)\subseteq A$. 
\end{definition}
\begin{definition}{Closed Set}{}\gline{}
  A set $A\subseteq \C$ is \ul{closed} if its complement is open.
  \begin{theorem}{Closet Set Cont.}{}\gline{}
    $A$ is closed if and only if $A$ contains all of its boundary points. 
  \end{theorem}
  \textit{Note:} $\C$ and $\emptyset$ are the only sets that are both open and closed. 
\end{definition}
\begin{definition}{Subsets of $A$}{}\gline{}
  For a set $A\subseteq \C$,
  \begin{itemize}
    \item The \ul{interior of $A$} \texttt{int}($A$), or $\mathring{A}$, is the set of all interior points of $A$
    \item The \ul{exterior of $A$}, \texttt{ext}($A$) is the set of all exterior points of $A$
    \item The \ul{boundary of $A$}, $\partial(A)$, is the set of all boundary points of $A$.
    \item The \ul{closure of $A$}, \texttt{cl}($A$) or $\bar A$, is $A\cup \partial(A)$. 
  \end{itemize}
  \begin{theorem}{Interior of $A$ cont.}{}\gline{}
    The $\texttt{int}(A)$ is the largest open set contained in $A$
  \end{theorem}
  \begin{theorem}{Closure of $A$ cont.}{}\gline{}
     $\bar A$ is the smallest closed set contained in $A$
  \end{theorem}
\end{definition}
\begin{definition}{Connected}{}\gline{}
  An open set $A\subseteq \C$ is \ul{connected} (or \ul{path-connected}) if every pari of points in $A$ can be joined by a polygonal path fully contained in $A$.
\end{definition}
\begin{definition}{Domain}{}\gline{}
  A \ul{domain} is a nonempty open connected set. A \ul{region} is a domain together with some or none of its boundary points.\gline{}
  \textit{Convention: }For any domain, any region created from it will be considered connected. 
\end{definition}
\begin{definition}{Bounded}{}\gline{}
  A set $A\subseteq \C$ is \ul{bounded} if there exists $R > 0$ such that $|z|<R$ for all $z\in \C$. If a set if not bounded, we say it is $\ul{unbounded}$. 
\end{definition}
\begin{definition}{Compact}{}\gline{}
  A set $A\subseteq \C$ is \ul{compact} if it is closed and bounded. 
\end{definition}
\begin{theorem}{}{}\gline{}
  Suppose $u(x,y)$ is a real-valued function defined on a domain $D$. 
  \begin{align*}
    &\text{If }\frac{\partial u}{\partial x}=\frac{\partial u}{\partial y}=0\\
    &\text{on all of }D\text{, then }u\text{ is constant of }D.
  \end{align*}
  \begin{example}
    Suppose $u(x,y)$ satisfies $\frac{\partial u}{\partial y}= x$ and $\frac{\partial u}{\partial x} = y$ on a domain $D$. Prove that $u(x,y) = xy+c$ for some constant $c$. 
    \begin{proof}
      Define $v(x,y) = u(x,y)-xy$ on $D$.
      \begin{align*}
        \text{Then, }&\frac{\partial v}{\partial x}=\frac{\partial u}{\partial x}-y = y-y = 0\text{, and}\\
        &\frac{\partial v}{\partial y}=\frac{\partial u}{\partial y}-x = x-x = 0.
      \end{align*} 
      So $v(x,y) = c$ for some constant $c$. Hence $u(x,y) = xy+c$ on $D$.
    \end{proof}
  \end{example}
\end{theorem}

\subsection*{2.2-1 Limits of Sequences and Functions}
\begin{definition}{Sequence of Complex Numbers}{}\gline{}
  A sequence of complex numbers $\{z_n\}^\infty_{n=1}$ \ul{converges to $z_0\in\C$} if for every $\varepsilon>0$, there exists $N\in\N$ such that if $n>N$, then $|z_n-z_0|<0$. 
\end{definition}
We say that:
\begin{itemize}
  \item $z_n$ converges to $z_0$
  \item $z_0$ is the limit of $z_n$
  \item $\lim_{n\to\infty}z_n = z_0$
  \item $z_n\to z_0$ as $n\to\infty$
\end{itemize}
Take as a give the following properties:
\begin{itemize}
  \item $\lim_{n\to\infty}\frac{1}n = 0$
  \item $\displaystyle \lim_{n\to\infty}\frac{an^k + \text{ lower power stuff}}{bn^k + \text{ lower power stuff}} = \frac{a}b$
  \item If $|z|<1,\quad \lim_{n\to\infty}z^n = 0$
  \item If $\{z_n\}_{n=1}^\infty$ does not converge to any $z_0\in\C$, we say $\{z_n\}^\infty_{n=1}$ \ul{diverges}.
\end{itemize}
\begin{definition}{Limit of a Function}{}\gline{}
  Let $f$ be defined on a deleted neighborhood of $z_0$. Then, $\lim_{z\to z_0}w_0\in\C$ if for every $\varepsilon > 0$, there exists $\delta >0$ such that if $ 0 < |z- z_0|<\delta$, then 
  \[|f(z) - w_0|<\varepsilon\]
\end{definition}
\begin{theorem}{}{}\gline{}
  Sequence limits and function limits are unique.\gline{}
  \textbf{Fact: }If you can plug $z_0$ info $f$ and get an unambiguous answer, then $\lim_{z\to z_0}f(z)=f(z_0)$. 
  \begin{corollary}{Limit Laws}{}\gline{}
    Suppose $\lim_{z\to z_0}f(z) = A\in |C$ and $\lim_{z\to z_0}g(z) = B\in\C$. Then, 
    \begin{itemize}
      \item $\lim_{z\to z_0}(f(z)\pm g(z)) = A\pm B$;
      \item $\lim_{z\to z_0}(f(z)\cdot g(z))=A\cdot B$;
      \item $\displaystyle\lim_{z\to z_0}\frac{f(z)}{g(z)}=\frac{A}B\quad \text{ if } B\ne A$. 
    \end{itemize}
  \end{corollary}
\end{theorem}
Here is some pretty helpful limit laws that would be pretty good to know:
\begin{theorem}{Limit Shit}{}\gline{}
  $|z_n|\to 0$ as $n\to\infty$ iff $z_n\to 0$ as $n\to\infty$.
  \begin{enumerate}
    \item \begin{enumerate}
      \item [(a)]$\displaystyle\left\{\cpr{\frac{i}2}^n\right\}_{n=1}^\infty \Rightarrow \left\{\frac{i}2, -\frac{1}{4}, -\frac{i}8, \frac{1}{16},\cdots\right\}$
      \item [(b)]$\displaystyle\left\{(2i)^n\right\}_{n=1}^\infty \Rightarrow \{2i, -4, -8i, 16, 32i\}$
    \end{enumerate}
    \item \begin{enumerate}
      \item [(c)] $\displaystyle \lim_{z\to 0}\frac{1}z = \infty$.
    \end{enumerate}
    \item \begin{enumerate}
      \item [(d)] $f(x+iy) = \frac{x^2y}{x^4 + y^4} \Rightarrow\;$does not converge. 
    \end{enumerate}
  \end{enumerate}
\end{theorem}
