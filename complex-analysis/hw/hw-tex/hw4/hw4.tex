			(1-i)^2 &= 2i \\

\item Suppose $f(z)$ is continuous at $z_0$. Prove that $\overline{f(z)}$,  $\Re(z)$, and $\Im(z)$ are all continuous at $z_0$ by writing each one as a combination of continuous functions.
\begin{proof}
	First, let $\delta >0$ and $\varepsilon> 0$ such that $|z-z_0|<\delta$ and $|f(z)-f(z_0)|<\varepsilon$. \gline{}
		Via the properties of the complex conjugate, we know that:
		\begin{align*}
			|f(z)-f(z-0)|&=\overline{|f(z)-f(z_0)|}\\
			&=|\overline{f(z)}-\overline{f(z_0)}|<\varepsilon.
		\end{align*}
		Thus, $\overline{f(z_0)}$ is continuous. \gline{}
		For $\Re(z)$ and $\Im(z)$, consider that:
		\begin{align*}
			\Re(z) = \frac{z + \bar z}2 &\quad\Im(z) = \frac{z-\bar z}{2}
		\end{align*}
		Thus, because both the real and imaginary function are compositions of other continuous functions, they themselves are continuous. 
\end{proof}

		Let $\varepsilon> 0$ and let $\delta  = \varepsilon\sqrt{u_0}$. Let $u\in\{x\in\R : x\ge 0\}$ such that $|u - u_0|<\delta$. Thus, we have that:
		\begin{align*}

		Thus, the square root function is continuous. \gline{}
