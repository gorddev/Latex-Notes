%%%%%%%%%% Title Formats

% SECTION
\titleformat{\section}
  {\normalfont\Large\bfseries}{}{0pt}{
    \begin{tcolorbox}[sectionbox]#1\end{tcolorbox}
  }[\vspace{-1.55cm}
  \addcontentsline{toc}{section}{#1}]

% SUBSECTION
\titleformat{\subsection}
{\normalfont\large\bfseries}{}{0pt}{
\begin{tcolorbox}[subsectionbox]#1\end{tcolorbox}
}[\vspace{-0.8cm}\addcontentsline{toc}{subsection}{#1}]


%%%%%%%%%% Custom Environments

% PROBLEM
\renewenvironment{proof}[1]
{\vspace{-0.1cm}\begin{tcolorbox}[proofbox]
  \textit{Proof. }#1}
{\qed\end{tcolorbox}\vspace{0.3em}}

% PROBLEM
\newenvironment{lemma}[2]
{\vspace{-0.1cm}\begin{tcolorbox}[lemmabox]
  \textit{Lemma #1. }#2}
{\end{tcolorbox}\vspace{0.3em}}

% PROBLEM
\newenvironment{problem}[1]
{\vspace{-0.05cm}\begin{tcolorbox}[problembox]
  #1}
{\end{tcolorbox}\vspace{0.3em}}

% REFLECTION
\newenvironment{reflection}[1]
{\sffamily\vspace{-0.05cm}\begin{tcolorbox}[reflectionbox]
  #1}
{\end{tcolorbox}\vspace{0.3em}}

% GRADE
\newenvironment{grade}[1]
{\sffamily\vspace{-0.05cm}\begin{tcolorbox}[gradebox]
  \textbf{Grade: }#1}
{\end{tcolorbox}\vspace{0.3em}}

% IMAGE
\newcommand{\gimage}[2]{
  \begin{center}
  \gimagebox{\includegraphics[width=#2]{#1}}
  \end{center}
}
